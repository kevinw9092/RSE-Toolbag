# RSE-Toolbag settings
# Change them in Esc > MODS if you use RSE-ModMenu, or edit this file with the game closed.

# ---- Toolbag storage (restart the game after a change) ----
# off     = no toolbag slots (default). The tool key, prompt and auto tool still work from your bag.
# private = a hidden fifth inventory tab of its own type: pickups never go there. EXPERIMENTAL.
# items   = a hidden fifth tab of the bag's type: accepts every tool, but pickups can overflow into it.
# Try a new mode on a throwaway character first (see README). Before switching back to off,
# or removing the mod, click "Take all out" in the toolbag window.
ToolbagMode = off

# ---- Toolbag keys ----
# Modifier for 1-0 that equips toolbag slots 1-10: ALT (default; CTRL is dodge and SHIFT is sprint),
# SHIFT, CTRL or NONE (off).
SlotKeys = ALT

# ---- Tool key ----
# Face a rock, tree, vine, farm plot, kebbit burrow, buried treasure or fishing spot and press this key: the best tool for it comes out,
# from your toolbag first, then your bag. "none" turns it off. Restart after a change.
ToolKey = X

# ---- Quick row ----
# true = show the 10 toolbag slots above the hotbar, with their tools and the SHIFT + number keys.
QuickRow = true
# Where the row sits: Above health, Below health, or Above action bar (the 1-8 shortcut bar;
# only toolbag slots 1-8 show there, to line up with it).
QuickRowPosition = Above health
# Move that row up (+) or down (-), in UI units, -300 to 300.
QuickRowOffset = 0

# true = show "Switch Tool [X]" while you face something your carried tool can work on.
EquipPrompt = true

# How close a rock or tree must be, in meters (it must also be in front of you). 1.5 to 6.
ToolReach = 3.0

# Cuttable vines (thorny vines across entrances, vines over wells) need this tool.
# axe by default; pickaxe, spade, water, compost, rod or net if the game wants another.
VineTool = axe

# Farm plots: also take out the compost bucket once a plot is watered.
UseCompost = true

# ---- Auto tool ----
# true = hit a rock, tree or farm plot with the wrong item and the right tool comes out.
AutoTool = true
# true = also swap away from a melee weapon, but only for a rock or tree right in front of you.
AutoToolFromWeapon = true

# ---- Other ----
# true = picked-up items go into your bag instead of filling empty hotbar slots.
BagFirst = false

# true = extra lines in ue4ss\UE4SS.log (useful for bug reports).
Debug = false
